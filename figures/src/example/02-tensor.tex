% example, page 02: a tensor diagram with tikz-tensors -- the expansion
% psi(r1, r2) = sum_ij C_ij phi_i(r1) phi_j(r2): the coefficient array C (one
% entry across both sites, indices i and j) on the basis functions phi
% (circles), which carry the positions (wavy legs, because they are
% continuous). Nothing in it is a length or a coordinate: a stack of two sites
% and two layers, connected, with the lower layer's indices opened downward.
\documentclass[border=4pt]{standalone}
\input{preamble}
\usepackage{tikz-tensors}
\input{notation}

\begin{document}
\begin{tikzpicture}
  \tnstack{X}{2}{array, basis}
  \tnlayer{X}{array}{coefwide/$C$/2}
  \tnlayer{X}{basis}{2*fn/$\varphi$}
  \tnconnect[apart=basis]{X}
  \tnopen[edge=cont, label=$\mathbf r_{#1}$]{down}{X}
  \tnmid[left]{X-array-1-slot}{X-basis-1-slot}{$i$}
  \tnmid[right]{X-array-2-slot}{X-basis-2-slot}{$j$}
\end{tikzpicture}
\end{document}
